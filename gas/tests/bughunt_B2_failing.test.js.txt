const test = require('node:test');
const assert = require('node:assert/strict');
const C = require('../pure_comment.js');
test('restoreComment stores target_id trimmed like appendComment does', () => {
  const rs = C.restoreComment([], { id: 'CMT-00000003', target_id: ' PBI-1 ', author: 'a', created_at: '2026-10-01 00:00:00', body: 'x' });
  assert.equal(rs.ok, true);
  assert.equal(rs.rows[0].target_id, 'PBI-1');
});
