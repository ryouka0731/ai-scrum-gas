'use strict';
// バグ探し 2巡目 P: 規模・性能の「現状では落ちる」期待。通常の `npm test` には入らない（拡張子 .txt）。
// 走らせ方（worktree のルートで）:
//   cp gas/tests/bughunt2_P_failing.test.js.txt gas/tests/bughunt2_P_failing_run.test.js \
//     && node --test gas/tests/bughunt2_P_failing_run.test.js; rm gas/tests/bughunt2_P_failing_run.test.js
// 各テストは「実際に起きる大きさ」で、期待と現状の数字をメッセージに出す。

const test = require('node:test');
const assert = require('node:assert/strict');
const S = require('./bughunt2_P_support.js');
const { createHarness } = require('./kanban_harness.js');

const NEW_PBI = { title: 'n', description: '', acceptance_criteria: '', status: 'New', priority: 'Medium', size: '', sprint: '' };

// ---------------------------------------------------------------------------
// BUG-P2 (medium): コメントが「全部」すべてのビュー応答とコメント書き込みの応答に載る
// ---------------------------------------------------------------------------
// spec 第6段階は「履歴は増え続けるので、ビューの応答には載せない」とした。コメントも増え続けるのに
// 盤面・一覧・障害物の応答と、コメントを1件足すたびの応答（apiAddComment / Delete / Restore）に全対象・全件が載る。
// 6,000件（8人で1年ぶん。本文200文字）で、盤面を開くたび・コメントを送るたびに約 4.5MB（UTF-8）を往復する。
// 期待: ビュー応答の大きさはコメント数にほぼ依らない（件数だけ、または開いた対象の分だけ）。
test('BUG-P2a. コメント 6,000 件で、盤面の応答が肥大する（コメント無しの 50 倍以上）', () => {
  const none = S.createCtx(S.standardFiles(300, null));
  const many = S.createCtx(S.standardFiles(300, 6000, { bodyChars: 200 }));
  const base = S.payloadBytes(none.ctx.apiGetView('board'));
  const big = S.payloadBytes(many.ctx.apiGetView('board'));
  assert.ok(big < 3 * base, '盤面の応答 ' + base + ' バイト → ' + big + ' バイト（' + (big / base).toFixed(1) + ' 倍）。コメントの本文が全部載っている');
});

test('BUG-P2b. コメントを1件送るたびの応答に、全コメントが載る（6,000件で約 4.5MB）', () => {
  const c = S.createCtx(S.standardFiles(300, 6000, { bodyChars: 200 }));
  const r = c.ctx.apiAddComment(S.pid(1), 'こんにちは');
  assert.equal(r.ok, true);
  const bytes = S.payloadBytes(r);
  assert.ok(bytes < 200000, '書き込みの応答が ' + bytes + ' バイト（送った1件は約 300 バイト）');
});

// ---------------------------------------------------------------------------
// BUG-P4 (medium): 履歴パネルの DOM が、描く行数の予算なしに 20万ノードになる
// ---------------------------------------------------------------------------
// kanban.html の HISTORY_DIFF_BUDGET（LCS の表 200万セル）は CPU だけを縛る。予算を使い切った後は、
// 前・後を丸ごと del/add の行として出す（1件 = 前の行数 + 後の行数 の <div>）。500行の説明を200回直した履歴は
// 200 × 1,000 行 = 約20万の <div> になり、実ブラウザでは数秒の固まりとスクロールの重さになる。
// 期待: 1回の描画で作る DOM は一定の上限（ここでは 50,000 ノード）で頭打ちになる（折りたたむ・省略する）。
test('BUG-P4. 履歴 200 件 × 500 行の説明で、履歴パネルの DOM ノードが 20 万になる', () => {
  const cols = [{ status: 'New', cards: [{ id: 'PBI-001', title: 'A', updated_at: 'T1' }] },
    { status: 'Ready', cards: [] }, { status: 'In Progress', cards: [] }, { status: 'Review', cards: [] }, { status: 'Done', cards: [] }];
  const h = createHarness(cols);
  h.sandbox.load();
  h.calls[0].handlers.success({ ok: true, name: 'board', view: { columns: cols }, summary: { byStatus: [], total: { count: 1, points: 0 } }, comments: {}, sprintChoices: [] });
  h.openCard('PBI-001');
  h.historyToggle('panel-history');
  const entries = [];
  for (let i = 0; i < 200; i++) {
    const a = []; const b = [];
    for (let k = 0; k < 500; k++) { a.push('行' + k + ' v' + i); b.push('行' + k + ' v' + (i + 1)); }
    entries.push({ at: '2026-01-01 00:' + String(Math.floor(i / 60)).padStart(2, '0') + ':' + String(i % 60).padStart(2, '0'),
      actor: 'a@example.com', action: 'update', field: 'description', before: a.join('\n'), after: b.join('\n') });
  }
  h.calls[h.calls.length - 1].handlers.success({ ok: true, entries: entries });
  let nodes = 0;
  (function walk(e) { nodes++; (e.children || []).forEach(walk); })(h.sandbox.document.getElementById('panel-history'));
  assert.ok(nodes <= 50000, '履歴パネルの DOM ノード ' + nodes);
});
